// --- HELLBOARD 3D ENGINE --- let scene, camera, renderer; let player, boardMesh; let obstacles = []; let terrainSegments = []; let particles; // Game State let gameState = 'MENU'; // MENU, PLAYING, GAMEOVER let score = 0; let distance = 0; let bestScore = localStorage.getItem('hellboard_best') || 0; // Physics & Controls let playerX = 0; let targetX = 0; let velocityY = 0; let isJumping = false; let isBraking = false; let currentSpeed = 0.7; let difficultyMultiplier = 1; // Controls tracking let touchStartX = 0; let touchStartY = 0; let keys = { left: false, right: false, up: false, down: false }; const LANE_WIDTH = 2.2; const MAX_LANE = 2.5; const SEGMENT_LENGTH = 60; const TOTAL_SEGMENTS = 8; document.getElementById('best-score').innerText = `${bestScore} pts`; function init() { // 1. Scene Setup scene = new THREE.Scene(); scene.background = new THREE.Color(0x050510); scene.fog = new THREE.FogExp2(0x0a0a20, 0.012); // 2. Camera Setup camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000); camera.position.set(0, 4.5, 7); camera.lookAt(0, 1, -2); // 3. Renderer Setup renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: "high-performance" }); renderer.setSize(window.innerWidth, window.innerHeight); renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)); renderer.shadowMap.enabled = true; document.getElementById('canvas-container').appendChild(renderer.domElement); // 4. Lighting const ambientLight = new THREE.AmbientLight(0xddeeff, 0.7); scene.add(ambientLight); const dirLight = new THREE.DirectionalLight(0xffeedd, 1.2); dirLight.position.set(10, 20, 10); dirLight.castShadow = true; scene.add(dirLight); const hemLight = new THREE.HemisphereLight(0x38bdf8, 0xf97316, 0.6); scene.add(hemLight); // 5. Build World & Player createTerrain(); createPlayer(); createSnowParticles(); // 6. Event Listeners window.addEventListener('resize', onWindowResize); setupControls(); // Start Loop animate(); } // --- PLAYER SETUP --- function createPlayer() { const playerGroup = new THREE.Group(); // Snowboard const boardGeo = new THREE.BoxGeometry(0.8, 0.08, 2.2); const boardMat = new THREE.MeshStandardMaterial({ color: 0xff3300, roughness: 0.3, metalness: 0.8 }); boardMesh = new THREE.Mesh(boardGeo, boardMat); boardMesh.position.y = 0.04; boardMesh.castShadow = true; playerGroup.add(boardMesh); // Body / Rider const bodyGeo = new THREE.BoxGeometry(0.5, 0.9, 0.4); const bodyMat = new THREE.MeshStandardMaterial({ color: 0x00f0ff, roughness: 0.4 }); const body = new THREE.Mesh(bodyGeo, bodyMat); body.position.y = 0.6; body.castShadow = true; playerGroup.add(body); // Head const headGeo = new THREE.SphereGeometry(0.25, 16, 16); const headMat = new THREE.MeshStandardMaterial({ color: 0xffdbac }); const head = new THREE.Mesh(headGeo, headMat); head.position.y = 1.25; head.castShadow = true; playerGroup.add(head); player = playerGroup; player.position.set(0, 0, 0); scene.add(player); } // --- INFINITE TERRAIN GENERATION --- function createTerrain() { for (let i = 0; i < TOTAL_SEGMENTS; i++) { const seg = generateTerrainSegment(-i * SEGMENT_LENGTH, i); terrainSegments.push(seg); scene.add(seg); } } function generateTerrainSegment(zPos, index) { const group = new THREE.Group(); group.position.z = zPos; group.userData = { segmentIndex: index }; // Slope Base const geo = new THREE.PlaneGeometry(30, SEGMENT_LENGTH, 16, 32); geo.rotateX(-Math.PI / 2); const mat = new THREE.MeshStandardMaterial({ color: index > 3 ? 0x1a0d00 : 0xe2e8f0, roughness: 0.8, flatShading: true }); const mesh = new THREE.Mesh(geo, mat); mesh.receiveShadow = true; group.add(mesh); // Populate Obstacles (skip first 2 segments for safety start) if (index > 1) { populateSegmentObstacles(group); } return group; } function populateSegmentObstacles(group) { // Increase density continuously based on difficulty multiplier const count = Math.floor(4 + (difficultyMultiplier * 2) + Math.random() * 3); for (let j = 0; j < count; j++) { const obs = createObstacleOrBonus((Math.random() - 0.5) * (SEGMENT_LENGTH - 10)); group.add(obs); obstacles.push(obs); } } function createObstacleOrBonus(localZ) { const typeRand = Math.random(); const xPos = (Math.random() - 0.5) * (LANE_WIDTH * 4.2); const container = new THREE.Group(); container.position.set(xPos, 0, localZ); if (typeRand < 0.45) { // Pine Tree / Hell Spire const trunkGeo = new THREE.CylinderGeometry(0.15, 0.2, 1, 8); const trunkMat = new THREE.MeshStandardMaterial({ color: 0x5c4033 }); const trunk = new THREE.Mesh(trunkGeo, trunkMat); trunk.position.y = 0.5; trunk.castShadow = true; container.add(trunk); const leavesGeo = new THREE.ConeGeometry(0.9, 2.2, 8); const leavesMat = new THREE.MeshStandardMaterial({ color: difficultyMultiplier > 1.5 ? 0x450a0a : 0x0f172a, roughness: 0.9 }); const leaves = new THREE.Mesh(leavesGeo, leavesMat); leaves.position.y = 1.8; leaves.castShadow = true; container.add(leaves); container.userData = { type: 'tree' }; } else if (typeRand < 0.75) { // Lava Puddle / Rock const rockGeo = new THREE.DodecahedronGeometry(0.8, 1); const rockMat = new THREE.MeshStandardMaterial({ color: 0xf97316, emissive: 0xc2410c, emissiveIntensity: 0.7 }); const rock = new THREE.Mesh(rockGeo, rockMat); rock.position.y = 0.4; rock.castShadow = true; container.add(rock); container.userData = { type: 'lava' }; } else { // Booster Pad const padGeo = new THREE.BoxGeometry(1.5, 0.05, 1.5); const padMat = new THREE.MeshStandardMaterial({ color: 0x06b6d4, emissive: 0x0891b2, emissiveIntensity: 0.9 }); const pad = new THREE.Mesh(padGeo, padMat); pad.position.y = 0.03; container.add(pad); container.userData = { type: 'boost' }; } return container; } // --- PARTICLES --- function createSnowParticles() { const particleCount = 400; const geo = new THREE.BufferGeometry(); const positions = new Float32Array(particleCount * 3); for (let i = 0; i < particleCount * 3; i += 3) { positions[i] = (Math.random() - 0.5) * 30; positions[i + 1] = Math.random() * 15; positions[i + 2] = (Math.random() - 0.5) * 60; } geo.setAttribute('position', new THREE.BufferAttribute(positions, 3)); const mat = new THREE.PointsMaterial({ color: 0xffffff, size: 0.12, transparent: true, opacity: 0.7 }); particles = new THREE.Points(geo, mat); scene.add(particles); } function updateParticles() { const positions = particles.geometry.attributes.position.array; for (let i = 1; i < positions.length; i += 3) { positions[i] -= currentSpeed * 0.8; if (positions[i] < 0) positions[i] = 15; } particles.geometry.attributes.position.needsUpdate = true; } // --- CONTROLS --- function setupControls() { window.addEventListener('keydown', (e) => { if (e.key === 'ArrowLeft' || e.key === 'a') keys.left = true; if (e.key === 'ArrowRight' || e.key === 'd') keys.right = true; if (e.key === 'ArrowUp' || e.key === 'w') jump(); if (e.key === 'ArrowDown' || e.key === 's') isBraking = true; }); window.addEventListener('keyup', (e) => { if (e.key === 'ArrowLeft' || e.key === 'a') keys.left = false; if (e.key === 'ArrowRight' || e.key === 'd') keys.right = false; if (e.key === 'ArrowDown' || e.key === 's') isBraking = false; }); const bindBtn = (id, startAction, endAction) => { const el = document.getElementById(id); if (!el) return; el.addEventListener('touchstart', (e) => { e.preventDefault(); startAction(); }, { passive: false }); el.addEventListener('touchend', (e) => { e.preventDefault(); if(endAction) endAction(); }, { passive: false }); el.addEventListener('mousedown', (e) => { e.preventDefault(); startAction(); }); el.addEventListener('mouseup', (e) => { e.preventDefault(); if(endAction) endAction(); }); }; bindBtn('btn-left', () => keys.left = true, () => keys.left = false); bindBtn('btn-right', () => keys.right = true, () => keys.right = false); bindBtn('btn-jump', () => jump()); bindBtn('btn-brake', () => isBraking = true, () => isBraking = false); window.addEventListener('touchstart', (e) => { touchStartX = e.touches[0].clientX; touchStartY = e.touches[0].clientY; }, { passive: true }); window.addEventListener('touchend', (e) => { if (gameState !== 'PLAYING') return; const dx = e.changedTouches[0].clientX - touchStartX; const dy = e.changedTouches[0].clientY - touchStartY; if (Math.abs(dx) > Math.abs(dy)) { if (dx > 30) targetX = Math.min(MAX_LANE, targetX + LANE_WIDTH); else if (dx < -30) targetX = Math.max(-MAX_LANE, targetX - LANE_WIDTH); } else { if (dy < -30) jump(); else if (dy > 30) { isBraking = true; setTimeout(() => isBraking = false, 800); } } }, { passive: true }); } function jump() { if (!isJumping && gameState === 'PLAYING') { velocityY = 0.28; isJumping = true; } } // --- GAME LOGIC & LOOP --- function startGame() { gameState = 'PLAYING'; score = 0; distance = 0; currentSpeed = 0.7; difficultyMultiplier = 1; playerX = 0; targetX = 0; player.position.set(0, 0, 0); velocityY = 0; // Reset obstacles obstacles = []; terrainSegments.forEach((seg, index) => { seg.position.z = -index * SEGMENT_LENGTH; // Clear children except ground mesh (index 0) for (let i = seg.children.length - 1; i > 0; i--) { seg.remove(seg.children[i]); } if (index > 1) { populateSegmentObstacles(seg); } }); document.getElementById('menu-screen').classList.add('opacity-0', 'pointer-events-none'); document.getElementById('gameover-screen').classList.add('opacity-0', 'pointer-events-none'); document.getElementById('hud').classList.remove('opacity-0'); document.getElementById('touch-controls').classList.remove('opacity-0'); } function gameOver() { gameState = 'GAMEOVER'; if (score > bestScore) { bestScore = score; localStorage.setItem('hellboard_best', bestScore); } document.getElementById('final-score').innerText = score; document.getElementById('final-dist').innerText = `${Math.floor(distance)}m`; document.getElementById('final-best').innerText = bestScore; document.getElementById('gameover-screen').classList.remove('opacity-0', 'pointer-events-none'); document.getElementById('hud').classList.add('opacity-0'); document.getElementById('touch-controls').classList.add('opacity-0'); } function animate() { requestAnimationFrame(animate); if (gameState === 'PLAYING') { // 1. Progressive Difficulty Scaling (Gets strictly harder and faster forever) distance += currentSpeed * 0.6; score += Math.floor(currentSpeed * 3); // Difficulty multiplier climbs continuously without limit difficultyMultiplier = 1 + (distance / 400); let targetSpeed = 0.7 + (distance / 350); if (isBraking) { currentSpeed = Math.max(0.4, currentSpeed - 0.04); } else { currentSpeed += (targetSpeed - currentSpeed) * 0.08; } // HUD updates document.getElementById('score-val').innerText = score; document.getElementById('speed-multiplier-txt').innerText = currentSpeed.toFixed(1); // 2. Steering & Movement if (keys.left) targetX = Math.max(-MAX_LANE, targetX - 0.09); if (keys.right) targetX = Math.min(MAX_LANE, targetX + 0.09); playerX += (targetX - playerX) * 0.15; player.position.x = playerX; boardMesh.rotation.z = (targetX - playerX) * 0.4; // 3. Jump Physics if (isJumping) { player.position.y += velocityY; velocityY -= 0.015; if (player.position.y <= 0) { player.position.y = 0; isJumping = false; velocityY = 0; } } // 4. Infinite Terrain & Obstacle Recycling const totalSpan = TOTAL_SEGMENTS * SEGMENT_LENGTH; terrainSegments.forEach(seg => { seg.position.z += currentSpeed; // When segment moves behind camera, recycle it to the front if (seg.position.z > SEGMENT_LENGTH) { seg.position.z -= totalSpan; // Remove old obstacle references belonging to this segment for (let i = seg.children.length - 1; i > 0; i--) { const child = seg.children[i]; const idx = obstacles.indexOf(child); if (idx > -1) obstacles.splice(idx, 1); seg.remove(child); } // Re-populate with fresh, denser scaled obstacles populateSegmentObstacles(seg); } }); // 5. Obstacle Collision Detection & Cleanup for (let i = obstacles.length - 1; i >= 0; i--) { const obs = obstacles[i]; const worldPos = new THREE.Vector3(); obs.getWorldPosition(worldPos); // Clean up objects safely far behind player if (worldPos.z > 8) { obstacles.splice(i, 1); if (obs.parent) obs.parent.remove(obs); continue; } // Check collision const dx = player.position.x - worldPos.x; const dz = player.position.z - worldPos.z; const dist = Math.sqrt(dx * dx + dz * dz); if (dist < 1.1) { if (obs.userData.type === 'boost') { // Boost Pad! currentSpeed += 0.6; score += 300; if (obs.parent) obs.parent.remove(obs); obstacles.splice(i, 1); } else if (!isJumping || player.position.y < 0.4) { // Crash! gameOver(); } } } updateParticles(); } renderer.render(scene, camera); } function onWindowResize() { camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix(); renderer.setSize(window.innerWidth, window.innerHeight); } // Attach UI buttons document.getElementById('start-btn').addEventListener('click', startGame); document.getElementById('restart-btn').addEventListener('click', startGame); // Initialize Engine on load window.onload = init;
